import com.fasterxml.jackson.databind.ObjectMapper
import jakarta.annotation.PostConstruct
import org.springframework.beans.factory.annotation.Value
import org.springframework.core.io.ResourceLoader
import org.springframework.http.HttpStatus
import org.springframework.stereotype.Component
import org.springframework.web.bind.annotation.ResponseStatus
import java.io.File
import java.nio.file.Files
import java.nio.file.Paths
import java.util.concurrent.ConcurrentHashMap
import java.util.concurrent.atomic.AtomicLong

@ResponseStatus(HttpStatus.NOT_FOUND)
class ResourceNotFoundException(msg: String) : RuntimeException(msg)

@Component
class JsonDataLoader(
    private val mapper: ObjectMapper,
    private val resourceLoader: ResourceLoader,
    @Value("\${app.data.storage-dir:./data}") private val storageDir: String
) {
    fun <T> loadList(path: String, clazz: Class<T>): List<T> {
        val target = Paths.get(storageDir).resolve(path.trimStart('/'))
        if (Files.exists(target)) {
            val type = mapper.typeFactory.constructCollectionType(List::class.java, clazz)
            return mapper.readValue(target.toFile(), type)
        }
        val res = resourceLoader.getResource("classpath:$path")
        if (!res.exists()) return emptyList()
        val type = mapper.typeFactory.constructCollectionType(List::class.java, clazz)
        val data: List<T> = mapper.readValue(res.inputStream, type)
        persistList(path, data)
        return data
    }

    @Synchronized
    fun <T> persistList(path: String, data: List<T>) {
        val target = Paths.get(storageDir).resolve(path.trimStart('/'))
        target.parent?.let { Files.createDirectories(it) }
        mapper.writerWithDefaultPrettyPrinter().writeValue(target.toFile(), data)
    }
}

abstract class AbstractRepository<T : Any>(
    private val loader: JsonDataLoader,
    private val fileName: String,
    private val entityClass: Class<T>,
    private val idGetter: (T) -> Long?,
    private val idSetter: (T, Long) -> Unit
) {
    private val storage = ConcurrentHashMap<Long, T>()
    private val sequence = AtomicLong(0)

    @PostConstruct
    @Synchronized
    fun init() {
        val data = loader.loadList(fileName, entityClass)
        var maxId = 0L
        for (item in data) {
            val id = idGetter(item) ?: continue
            storage[id] = item
            maxId = maxOf(maxId, id)
        }
        sequence.set(maxId)
    }

    @Synchronized
    fun save(entity: T): T {
        var id = idGetter(entity)
        if (id == null || id <= 0) {
            id = sequence.incrementAndGet()
            idSetter(entity, id)
        } else {
            sequence.accumulateAndGet(id, ::maxOf)
        }
        storage[id] = entity
        loader.persistList(fileName, storage.values.toList())
        return entity
    }

    fun findById(id: Long): T? = storage[id]
    fun findAll(): List<T> = storage.values.toList()
}

@Repository
class BlogPostDB(loader: JsonDataLoader, @Value("\${app.data.posts}") file: String) :

@Repository
class BlogUserDB(loader: JsonDataLoader, @Value("\${app.data.users}") file: String) :
    AbstractRepository<BlogUser>(loader, file, BlogUser::class.java, { it.id }, { u, id -> u.id = id })

@Service
class BlogService(private val postDB: BlogPostDB, private val userDB: BlogUserDB) {
    fun getAllPosts(): List<BlogPost> = postDB.findAll()

    fun getPostById(id: Long): BlogPost =
        postDB.findById(id) ?: throw ResourceNotFoundException("No post found with id: $id")

    fun searchPosts(title: String): List<BlogPostData> =
        postDB.findAll()
            .filter { it.title?.contains(title, ignoreCase = true) == true }
            .map { BlogPostData.from(it) }

    fun createPost(dto: NewBlogPostDTO): BlogPostData {
        val post = BlogPost(userId = dto.userId, title = dto.title, content = dto.content)
        return BlogPostData.from(postDB.save(post))
    }

    fun updatePost(id: Long, dto: UpdateBlogPostDTO): BlogPostData {
        val existing = postDB.findById(id) ?: throw ResourceNotFoundException("No post found with id: $id")
        existing.title = dto.title
        existing.content = dto.content
        return BlogPostData.from(postDB.save(existing))
    }

    fun filterUsers(name: String?, user: String?, email: String?, city: String?, company: String?): List<UserDetails> =
        userDB.findAll()
            .filter { u -> name == null || u.name?.contains(name, true) == true }
            .filter { u -> user == null || u.username?.contains(user, true) == true }
            .filter { u -> email == null || u.email?.contains(email, true) == true }
            .filter { u -> city == null || u.address?.city?.contains(city, true) == true }
            .filter { u -> company == null || u.company?.name?.contains(company, true) == true }
            .map { UserDetails.from(it) }

    fun getUserById(id: Long): UserDetails =
        userDB.findById(id)?.let { UserDetails.from(it) }
            ?: throw ResourceNotFoundException("User with id $id not found")

    fun getUserPosts(id: Long): UserPosts {
        val user = userDB.findById(id) ?: throw ResourceNotFoundException("User with id $id not found")
        val posts = postDB.findAll().filter { it.userId == id }.map { BlogPostData.from(it) }
        return UserPosts(UserDetails.from(user), posts)
    }
}

class Store:
    def __init__(self, filename: str, seed_path: Path):
        self.filename = filename
        self.target_path = STORAGE_DIR / filename
        self.seed_path = seed_path
        self.data: Dict[int, dict] = {}
        self.seq = 0
        self._load()

    def _load(self):
        STORAGE_DIR.mkdir(parents=True, exist_ok=True)
        source = self.target_path if self.target_path.exists() else self.seed_path
        if source.exists():
            with open(source, "r", encoding="utf-8") as f:
                items = json.load(f)
                for it in items:
                    self.data[it["id"]] = it
                    if it["id"] > self.seq:
                        self.seq = it["id"]
            self._flush()

    def _flush(self):
        with open(self.target_path, "w", encoding="utf-8") as f:
            json.dump(list(self.data.values()), f, indent=2)

    def save(self, item: dict) -> dict:
        with lock:
            if not item.get("id") or item["id"] <= 0:
                self.seq += 1
                item["id"] = self.seq
            else:
                self.seq = max(self.seq, item["id"])
            self.data[item["id"]] = item
            self._flush()
            return item

@app.get("/api/v1/users", response_model=List[UserDetails])
def get_users(
    name: Optional[str] = Query(None),
    username: Optional[str] = Query(None),
    email: Optional[str] = Query(None),
    city: Optional[str] = Query(None),
    company: Optional[str] = Query(None)
):
    results = []
    for u in user_store.data.values():
        if name and name.lower() not in u.get("name", "").lower(): continue
        if username and username.lower() not in u.get("username", "").lower(): continue
        if email and email.lower() not in u.get("email", "").lower(): continue
        if city and city.lower() not in (u.get("address") or {}).get("city", "").lower(): continue
        if company and company.lower() not in (u.get("company") or {}).get("name", "").lower(): continue
        results.append(map_user_details(u))

type Storage struct {
mu       sync.RWMutex
posts    map[int64]BlogPost
users    map[int64]BlogUser
postSeq  int64
dataDir  string
}

func NewStorage() *Storage {
s := &Storage{
posts:   make(map[int64]BlogPost),
users:   make(map[int64]BlogUser),
dataDir: "./data",
}
_ = os.MkdirAll(s.dataDir, 0755)
s.load()
return s
}

func (s *Storage) load() {
loadJSON := func(filename, seed string, target interface{}) {
path := filepath.Join(s.dataDir, filename)
b, err := os.ReadFile(path)
if err != nil {
b, _ = os.ReadFile(seed)
}
if len(b) > 0 {
_ = json.Unmarshal(b, target)
}
}

var pList []BlogPost
loadJSON("posts.json", "posts.json", &pList)
for _, p := range pList {
s.posts[p.ID] = p
if p.ID > s.postSeq { s.postSeq = p.ID }
}

var uList []BlogUser
loadJSON("users.json", "users.json", &uList)
for _, u := range uList {
s.users[u.ID] = u
}
}

func (s *Storage) persistPosts() {
var list []BlogPost
for _, p := range s.posts { list = append(list, p) }
b, _ := json.MarshalIndent(list, "", "  ")
_ = os.WriteFile(filepath.Join(s.dataDir, "posts.json"), b, 0644)
}

func writeJSON(w http.ResponseWriter, status int, data interface{}) {
w.Header().Set("Content-Type", "application/json")
w.WriteHeader(status)
_ = json.NewEncoder(w).Encode(data)
}

func writeError(w http.ResponseWriter, status int, msg string) {
writeJSON(w, status, ErrorResponse{Error: fmt.Sprintf("%d %s", status, http.StatusText(status)), Message: msg})
}

func main() {
s := NewStorage()
mux := http.NewServeMux()

mux.HandleFunc("/api/v1/posts", func(w http.ResponseWriter, r *http.Request) {

s.mu.RLock()
defer s.mu.RUnlock()
list := make([]BlogPost, 0, len(s.posts))
for _, p := range s.posts { list = append(list, p) }
writeJSON(w, http.StatusOK, list)
})

writeError(w, http.StatusBadRequest, "Invalid search title length")
return
}
s.mu.RLock()
defer s.mu.RUnlock()
var out []BlogPostData
for _, p := range s.posts {
if strings.Contains(strings.ToLower(p.Title), strings.ToLower(title)) {
out = append(out, BlogPostData{ID: p.ID, Title: p.Title, Content: p.Content})
}
}
writeJSON(w, http.StatusOK, out)

import express, { Request, Response, NextFunction } from 'express';
import fs from 'fs';
import path from 'path';
import { z } from 'zod';

const app = express();
app.use(express.json());

const DATA_DIR = path.resolve('./data');
const POSTS_FILE = path.join(DATA_DIR, 'posts.json');
const USERS_FILE = path.join(DATA_DIR, 'users.json');

interface BlogPost {
  id: number;
  userId: number;
  title: string;
  content: string;
}

interface BlogUser {
  id: number;
  name: string;
  username: string;
  email: string;
  address?: { city?: string };
  company?: { name?: string };
  website?: string;
}

if (!fs.existsSync(DATA_DIR)) fs.mkdirSync(DATA_DIR, { recursive: true });

function initStore<T extends { id: number }>(targetFile: string, seedFile: string): Map<number, T> {
  const store = new Map<number, T>();
  const src = fs.existsSync(targetFile) ? targetFile : path.resolve(seedFile);
  if (fs.existsSync(src)) {
    const list: T[] = JSON.parse(fs.readFileSync(src, 'utf-8'));
    list.forEach(i => store.set(i.id, i));
    if (!fs.existsSync(targetFile)) {
      fs.writeFileSync(targetFile, JSON.stringify(list, null, 2));
    }
  }
  return store;
}

const posts = initStore<BlogPost>(POSTS_FILE, './posts.json');
const users = initStore<BlogUser>(USERS_FILE, './users.json');
let postSeq = Math.max(0, ...Array.from(posts.keys()));

const persistPosts = () => {
  fs.writeFileSync(POSTS_FILE, JSON.stringify(Array.from(posts.values()), null, 2));
};

const mapUserDetails = (u: BlogUser) => ({
  userId: u.id,
  username: u.username,
  city: u.address?.city ?? null,
  name: u.name,
  email: u.email,
  website: u.website ?? null,
});

// Zod Schemas
const newPostSchema = z.object({
  userId: z.number().int().positive(),
  title: z.string().min(5).max(80),
  content: z.string().min(50).max(1000)
});

const updatePostSchema = z.object({
  title: z.string().min(5).max(80),
  content: z.string().min(50).max(1000)
});

// Endpoints
app.get('/api/v1/posts', (_req: Request, res: Response) => {
  res.json(Array.from(posts.values()));
});

app.get('/api/v1/posts/:id(\\d+)', (req: Request, res: Response) => {
  const id = Number(req.params.id);
  const post = posts.get(id);
  if (!post) return res.status(404).json({ error: '404 NOT_FOUND', message: `No post found with id: ${id}` });
  res.json(post);
});

app.get('/api/v1/posts/search/:title', (req: Request, res: Response) => {
  const title = req.params.title.toLowerCase();
  const matched = Array.from(posts.values())
    .filter(p => p.title.toLowerCase().includes(title))
    .map(({ id, title, content }) => ({ id, title, content }));
  res.json(matched);
});

app.post('/api/v1/posts/create', (req: Request, res: Response) => {
  const body = newPostSchema.parse(req.body);
  const post: BlogPost = { id: ++postSeq, ...body };
  posts.set(post.id, post);
  persistPosts();
  res.json({ id: post.id, title: post.title, content: post.content });
});

app.put('/api/v1/posts/:id(\\d+)', (req: Request, res: Response) => {
  const id = Number(req.params.id);
  if (!posts.has(id)) return res.status(404).json({ error: '404 NOT_FOUND', message: `No post found with id: ${id}` });
  const body = updatePostSchema.parse(req.body);
  const existing = posts.get(id)!;
  existing.title = body.title;
  existing.content = body.content;
  persistPosts();
  res.json({ id: existing.id, title: existing.title, content: existing.content });
});

app.get('/api/v1/users', (req: Request, res: Response) => {
  const { name, username, email, city, company } = req.query as Record<string, string | undefined>;
  const list = Array.from(users.values())
    .filter(u => !name || u.name.toLowerCase().includes(name.toLowerCase()))
    .filter(u => !username || u.username.toLowerCase().includes(username.toLowerCase()))
    .filter(u => !email || u.email.toLowerCase().includes(email.toLowerCase()))
    .filter(u => !city || (u.address?.city ?? '').toLowerCase().includes(city.toLowerCase()))
    .filter(u => !company || (u.company?.name ?? '').toLowerCase().includes(company.toLowerCase()))
    .map(mapUserDetails);
  res.json(list);
});

app.get('/api/v1/users/:userId(\\d+)', (req: Request, res: Response) => {
  const userId = Number(req.params.userId);
  const user = users.get(userId);
  if (!user) return res.status(404).json({ error: '404 NOT_FOUND', message: `User with id ${userId} not found` });
  res.json(mapUserDetails(user));
});

app.get('/api/v1/users/:userId(\\d+)/posts', (req: Request, res: Response) => {
  const userId = Number(req.params.userId);
  const user = users.get(userId);
  if (!user) return res.status(404).json({ error: '404 NOT_FOUND', message: `User with id ${userId} not found` });
  const userPosts = Array.from(posts.values())
    .filter(p => p.userId === userId)
    .map(({ id, title, content }) => ({ id, title, content }));
  res.json({ author: mapUserDetails(user), posts: userPosts });
});

app.use((err: Error, _req: Request, res: Response, _next: NextFunction) => {
  res.status(400).json({ error: 'BAD_REQUEST', message: err.message });
});

app.listen(8099, () => console.log('Node blog service listening on :8099'));
